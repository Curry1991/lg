    function playOrPause()
    {
        $output = null;
        $retval = null;
        exec("luna-send -n 1 -f luna://com.webos.media/getActivePipelines '{subscribe: false}'", $output, $retval);
        if ($retval == 0) {
            // print_r($retval);
            // print_r($output);
            $output = implode('', $output);
            $output = json_decode($output, true);
            // print_r($output);
            foreach ($output as $pipeline) {
                if (array_key_exists('mediaState', $pipeline)) {
                    $mediaState = $pipeline['mediaState'];
                    logmsg("playOrPause: found media state: " . $mediaState);
                    if ($mediaState == "pause") {
                        // print_r($pipeline);
                        // play
                        return [
                            "action" => "replace",
                            "keycode" => 207
                        ];
                    } else {
                        // pause
                        return [
                            "action" => "replace",
                            "keycode" => 119
                        ];
                    }
                }
            }
        }
        logmsg("playOrPause: didnt found active pipeline");
        // return ["action" => "ignore"];
        // play
        return [
            "action" => "replace",
            "keycode" => 207
        ];
    }
