    func lunaSend(name string, args interface{}) string {
        data, err := json.Marshal(args)
        if err != nil {
            log.Printf("failed to marshal json %v", args)
            return ""
        }
        var out bytes.Buffer
        cmd := exec.Command("luna-send", "-d", "-n", "1", fmt.Sprintf("luna://%s", name), string(data))
        cmd.Stdout = &out
        cmd.Run()
        log.Print(out.String())
        return out.String()
    }
    func handleRequest(w http.ResponseWriter, r *http.Request) {
        log.Printf("%s %s", r.Method, r.URL.Path)
        commandName := r.URL.Path[1:]
        var body interface{}
        err := json.NewDecoder(r.Body).Decode(&body)
        if err != nil {
            http.Error(w, err.Error(), http.StatusBadRequest)
            return
        }
        log.Printf("calling lunaSend with %s and %v", commandName, body)
        rsp := lunaSend(commandName, body)
        w.Header().Add("content-type", "application/json")
        w.Write([]byte(rsp))
    }
    func main() {
        host := os.Getenv("HOST")
        port, err := strconv.Atoi(os.Getenv("PORT"))
        if err != nil {
            panic("PORT is required and must be an int")
        }
        username, err := user.Current()
        if err != nil {
            log.Fatal(err)
        }
        log.Printf("running as user id %s (%s)", username.Uid, username.Username)
        http.HandleFunc("/", handleRequest)
        log.Printf("listening on %s:%d", host, port)
        http.ListenAndServe(fmt.Sprintf("%s:%d", host, port), nil)
    }
