my silly quick solution was a dumb bash script:
    inactivateFor=0
    while true; do
            currentPowerState=`luna-send -n 1 'luna://com.webos.service.tvpower/power/getPowerState' '{"subscribe":false}'`
            if echo $currentPowerState | grep -q '"Screen Saver"'; then
                    nactivateFor=$((inactivateFor+=1))
                    echo "screensaver is active for $inactivateFor secs"
            fi
            if echo $currentPowerState | grep -q '"Active"'; then
                    inactivateFor=0
            fi
            if [ $inactivateFor -ge 20 ] && echo $currentPowerState | grep -q '"Screen Saver"' ; then
                    echo "deactivating screen.. after $inactivateFor secs"
                    luna-send -n 1 "luna://com.webos.service.tvpower/power/turnOffScreen" '{}'
                    inactivateFor=0
            fi
    #       echo $currentPowerState
            sleep 1
    done
bash turnoffscreen.sh &
