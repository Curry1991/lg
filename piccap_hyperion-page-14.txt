    static ms_EN_XC_DIP_DATA_FMT v4l2_fmt_2_ms_fmt(u32 v4l2_fmt)
    {
        ms_EN_XC_DIP_DATA_FMT ms_fmt;
        switch(v4l2_fmt)
        {
        case V4L2_PIX_FMT_NV12:
            ms_fmt = MS_DIP_DATA_FMT_YUV420_SEMI_PLANER;
            break;
        case V4L2_PIX_FMT_YUYV:
            ms_fmt = MS_DIP_DATA_FMT_YUV422;
            break;
        case V4L2_PIX_FMT_RGB32:
            ms_fmt = MS_DIP_DATA_FMT_ARGB8888;
            break;
        default:
            MLP_VT_TRACE("v4l2_fmt 0x%x", v4l2_fmt);
            ms_fmt = MS_DIP_DATA_FMT_MAX;
            break;
        }
        return ms_fmt;
    }
    const struct mlp_vt_fmt *mlp_vt_find_format(struct mlp_vt_priv *vp,
                              const u32 *pixelformat,
                              int index)
    {
        const struct mlp_vt_fmt *fmt, *def_fmt = NULL;
        unsigned int i;
        int id = 0;
        if (index >= (int)ARRAY_SIZE(mlp_vt_formats))
            return NULL;
        for (i = 0; i < ARRAY_SIZE(mlp_vt_formats); ++i) {
            fmt = &mlp_vt_formats[i];
            if (vp && !(vp->fmt_flags & fmt->flags))
                continue;
            if (pixelformat && fmt->fourcc == *pixelformat)
                return fmt;
            if (index == id)
                def_fmt = fmt;
            id++;
        }
        return def_fmt;
    }

chroma plane with weaved U and V values. NV21 is the same but with

satdx62 02/26/2024 6:15 PM
Hi, I have no major problems with green/yellow tones with the SK6812RGBW
NW LEDs. I was able to adjust fairly accurately so that the basic colors
(RGB) were exactly right on a white wall. The yellow is also displayed
correctly. LUT calibration on an externally hosted HyperHDR, e.g., on a
Raspberry Pi, is not a difficulty as the function was primarily
developed for external USB grabbers and not for flat buffers. Here on
the LG, however, it is very difficult to achieve higher accuracy due to
the lack of memory resources. In my opinion, you can only perform a live
calibration with the correct image material. A distinction must be made
between the classic HyperHDR calibration and the advanced calibration.
And it's best to first go to Remote control > Live calibration and then,
if everything is correct, enter everything under Image processing and
save everything in any case. Especially @Tim Cook User has a lot of
experience with LUT calibration. Maybe he has some tips/instructions for
you. Otherwise, you can contact @awawa-dev at
https://github.com/awawa-dev/HyperHDR/issues. (edited)
