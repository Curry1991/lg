    #!/bin/bash
    PATH="$PATH:/home/root/.bin/"
    if [[ "$1" == "watcher" ]]; then
        while true; do
            /home/root/webserver/webserver.py
            notify "Web server stopped."
            sleep 3
        done
    else
        ( $0 watcher & )
    fi
