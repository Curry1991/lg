here's a bash script I put in /var/lib/webosbrew/init.d with chmod +x so
it can run at start-up that makes quick start+ work with piccap. For
some reason the existing piccapautostart script does not work well for
me
    SUSPEND="Prepare Suspend"
    RESUME="Screen On"
    echo "Starting Hyperion" & luna-send -n 1 -f 'luna://org.webosbrew.piccap.service/start' '{}'
    luna-send -i 'luna://com.webos.service.tvpower/power/getPowerState' '{"subscribe":true}' | while read ln; do echo "processing $ln"; if  echo "$ln" | grep -q "$RESUME"; then echo "Starting Hyperion" & luna-send -n 1 -f 'luna://org.webosbrew.piccap.service/start' '{}'; elif echo "$ln" | grep -q "$SUSPEND"; then echo "Stopping Hyperion" & luna-send -n 1 -f 'luna://org.webosbrew.piccap.service/stop' '{}'; fi; done
