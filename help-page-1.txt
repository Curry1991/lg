old tv:

    pid = 1324  # tvservice PID
    fd = os.open("/proc/%d/mem" % pid, os.O_RDWR)
    # 1-byte: movs r0,#0 -> movs r0,#1 at 0xc9ef00
    os.lseek(fd, 0xc9ef00, os.SEEK_SET)
    os.write(fd, "\x01\x20")
    os.close(fd)
    print "Done"
