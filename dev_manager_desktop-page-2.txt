    #!/usr/bin/env python
    from pywebostv.discovery import *    # Because I'm lazy, don't do this.
    from pywebostv.connection import *
    from pywebostv.controls import *
    from PyQt6.QtWidgets import QApplication, QWidget, QGridLayout, QPushButton
    import sys
    class RemoteControl(QWidget):
        def __init__(self):
            super().__init__()
            store = {'client_key': 'my_key'}
            client = WebOSClient("192.168.1.213")
            client.connect()
            for status in client.register(store):
                if status == WebOSClient.PROMPTED:
                    print("Please accept the connect on the TV!")
                elif status == WebOSClient.REGISTERED:
                    print("Registration successful!")
            inp = InputControl(client)
            inp.connect_input()
            self.setWindowTitle("LG TV Remote Control")
            self.setGeometry(100, 100, 300, 500)
            layout = QGridLayout()
            self.setLayout(layout)
            # Define the buttons and their positions in the grid
            button_positions = {
                "Menu": (0, 0),
                "Home": (0, 1),
                "Back": (0, 2),
                "Up": (1, 1),
                "Left": (2, 0),
                "Ok": (2, 1),
                "Right": (2, 2),
                "Down": (3, 1),
                "Exit": (4, 0),
                "Mute": (4, 2),
                "Volume Down": (5, 0),
                "Volume Up": (5, 2),
                "Channel Down": (6, 0),
                "Channel Up": (6, 2),
                "Play": (7, 0),
                "Pause": (7, 1),
                "Stop": (7, 2),
                "Rewind": (8, 0),
                "Fast Forward": (8, 2),
            }
            # Create buttons manually based on the grid positions and commands
            buttons = {
                "Menu": inp.menu,
                "Home": inp.home,
                "Back": inp.back,
                "Up": inp.up,
                "Left": inp.left,
                "Ok": inp.ok,
                "Right": inp.right,
                "Down": inp.down,
                "Exit": inp.exit,
                "Mute": inp.mute,
                "Volume Down": inp.volume_down,
                "Volume Up": inp.volume_up,
                "Channel Down": inp.channel_down,
                "Channel Up": inp.channel_up,
                "Play": inp.play,
                "Pause": inp.pause,
                "Stop": inp.stop,
                "Rewind": inp.rewind,
                "Fast Forward": inp.fastforward,
            }
            # Add buttons to the layout
            for label, (row, col) in button_positions.items():
                button = QPushButton(label)
                button.clicked.connect(lambda _, cmd=buttons[label]: self.run_command(cmd))
                layout.addWidget(button, row, col)
        def run_command(self, command):
            command()
    def main():
        app = QApplication(sys.argv)
        window = RemoteControl()
        window.show()
        sys.exit(app.exec())
    if __name__ == "__main__":
        main()
