        /// <summary>
        /// Set bias for pads that have a direction bit (PUPD) plus a two-bit resistor
        /// ladder (R0, R1). The PUPD bit selects up (0) vs down (1); R0/R1 encode the
        /// pull impedance (75k/10k/5k/2k). "None" writes the up direction but leaves
        /// R0/R1 per <c>resistance</c> (kept for compatibility with the original path).
        /// Only this topology can set a resistance.
        /// </summary>
        private void SetBiasR1R0(PullConfig reg, PullBias bias, ResistorValue res)
        {
            var b = _profile.IoRegions[(int)reg.Region];
            int pv = bias == PullBias.PullUp ? 0 : (bias == PullBias.PullDown ? 1 : 0);
            var (r0, r1) = ResToR1R0(res);
            _mmio.WriteBit(b + reg.PupdOffset, reg.PupdBit, pv);
            _mmio.WriteBit(b + reg.R0Offset, reg.R0Bit, r0);
            _mmio.WriteBit(b + reg.R1Offset, reg.R1Bit, r1);
        }
        /// <summary>
        /// Set bias for pads with two independent pull enable bits (PU = pull-up,
        /// PD = pull-down). Exactly one is asserted at a time: up = {PU 1, PD 0},
        /// down = {PU 0, PD 1}, none = {PU 0, PD 0}. No resistor control.
        /// </summary>
        private void SetBiasPuPd(PullConfig reg, PullBias bias)
        {
            var b = _profile.IoRegions[(int)reg.Region];
            var pa = b + reg.PuOffset;
            var da = b + reg.PdOffset;
            if (bias == PullBias.PullUp)
            {
                _mmio.WriteBit(pa, reg.PuBit, 1);
                _mmio.WriteBit(da, reg.PdBbit, 0);
            }
            else if (bias == PullBias.PullDown)
            {
                _mmio.WriteBit(pa, reg.PuBit, 0);
                _mmio.WriteBit(da, reg.PdBbit, 1);
            }
            else
            {
                _mmio.WriteBit(pa, reg.PuBit, 0);
                _mmio.WriteBit(da, reg.PdBbit, 0);
            }
