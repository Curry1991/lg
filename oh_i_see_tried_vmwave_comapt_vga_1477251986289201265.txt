Found the way for webos 5:
    gcc -m32 -static -o rootsh rootsh.c
    mkdir -p /mnt/webos
    qemu-nbd --connect=/dev/nbd0 Emulator-qemu/LG_webOS_TV_Emulator-5.0.0.qcow2
    mount /dev/nbd0p2 /mnt/webos
    cp rootsh /mnt/webos/var/rootsh
    chmod 4755 /mnt/webos/var/rootsh
    umount /mnt/webos
    qemu-nbd --disconnect /dev/nbd0
    ssh -i ./Emulator-qemu/webos-dev.key -p 5022 developer@localhost
    /var/rootsh
    mount -o remount,rw /etc
    passwd
    vi /lib/systemd/system/webos-rbd.target.wants/dropbear.service
    sync
