(signage)
    description "watchdog to reboot if tvservice dies"
    start on stopping tvservice
    env KLOG=/dev/kmsg
    # This job is started when either ls-hubd_public or ls-hubd_private
    # enter the 'stopping' state.  If this is part of a normal exit
    # (manual shutdown or reboot)
    #    RESULT will be "ok"
    #     - or -
    #   RUNLEVEL will be 0 or 1 (shutdown) or 6 (reboot)
    # both are testing since I did not always see the RESULT value set
    # since we only act in the 'else' case, the tests are negated
    # and anded.
    script
            echo "[TVSERVICE] TVService stopped..." > $KLOG
            echo "[TVSERVICE] $RESULT" > $KLOG
            echo "[TVSERVICE] $RUNLEVEL" > $KLOG
            if [ "$RESULT" != "ok" -a "$RUNLEVEL" != "0" -a "$RUNLEVEL" != "1" -a "$RUNLEVEL" != "6" ]; then
                    echo "[TVSERVICE] TVService stopped abnormally..." > $KLOG
                    /sbin/reboot
            fi
    end script
