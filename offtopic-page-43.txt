    import struct
    import sys
    import os
    HEADER_SIZE = 24
    LENGTH_OFFSET = 16
    MAX_REASONABLE_FRAME = 20000
    def decode_es(input_path, output_path):
        with open(input_path, "rb") as f:
            data = f.read()
        n = len(data)
        cursor = 0
        out = bytearray()
        frame_count = 0
        skipped_bytes = 0
        while cursor + HEADER_SIZE <= n:
            header = data[cursor:cursor + HEADER_SIZE]
            length = struct.unpack("<Q", header[LENGTH_OFFSET:LENGTH_OFFSET + 8])[0]
            if length == 0 or length > MAX_REASONABLE_FRAME or cursor + HEADER_SIZE + length > n:
                cursor += 1
                skipped_bytes += 1
                continue
            frame_start = cursor + HEADER_SIZE
            frame = data[frame_start:frame_start + length]
            if frame[0:1] != b"\xff":
                cursor += 1
                skipped_bytes += 1
                continue
            out += frame
            frame_count += 1
            cursor = frame_start + length
        with open(output_path, "wb") as f:
            f.write(out)
        print(f"Input:          {input_path} ({n:,} bytes)")
        print(f"Frames decoded: {frame_count:,}")
        if skipped_bytes:
            print(f"Bytes skipped while resyncing: {skipped_bytes:,}")
        print(f"Output:         {output_path} ({len(out):,} bytes)")
    def main():
        if len(sys.argv) < 2:
            print("give me es file")
            sys.exit(1)
        input_path = sys.argv[1]
        if len(sys.argv) >= 3:
            output_path = sys.argv[2]
        else:
            base, _ = os.path.splitext(input_path)
            output_path = base + ".mp3"
        decode_es(input_path, output_path)
    if __name__ == "__main__":
        main()
