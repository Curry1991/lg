        private function rpc_send(string $cmd, callable $body = null){
            $this->debug(" GET: {$cmd}");
            $length = pack('N', strlen($cmd));
            $buf = $length . $cmd;
            fwrite($this->sock, $buf);
            fflush($this->sock);
            if($body !== null){
                $body();
            }
            $hdr = fread($this->sock, 12);
            $this->debug("reply: " . bin2hex($hdr));
            $data = array_values(unpack('a4_a/N_b/N_c', $hdr));
            list($magic, $pkt_length, $body_length) = $data;
            $error = false;
            if($magic !== 'OK00' || $pkt_length < 12){
                $error = true;
            }
            if($error){
                throw new \Exception("ezinject protocol error: the server might have crashed");
            }
            $remaining = $pkt_length - $body_length - 12;
            $this->debug("OK, length: {$body_length}");
            if($body_length > 0){
                $data = fread($this->sock, $body_length);
            } else {
                $data = '';
            }
            fread($this->sock, $remaining);
            $this->debug(bin2hex($data));
            return $data;
        }
