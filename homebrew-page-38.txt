anyway, i use my own implementation:
    import { AsyncSink } from 'ix';
    type LunaTopicParams = Record<string, any>;
    export class Luna<T> extends AsyncSink<T> {
        private readonly bridge: PalmServiceBridge = new PalmServiceBridge();
        private readonly signal: AbortSignal;
        private constructor(uri: string, params: LunaTopicParams, signal?: AbortSignal) {
            super();
            this.handleAbort = this.handleAbort.bind(this);
            this.handleMessage = this.handleMessage.bind(this);
            this.signal = signal ?? new AbortController().signal;
            this.signal.addEventListener('abort', this.handleAbort);
            this.bridge.onservicecallback = this.handleMessage;
            this.bridge.call(uri, JSON.stringify(params));
        }
        public static async oneshot<T>(uri: string, params: LunaTopicParams = {}): Promise<T> {
            const controller = new AbortController();
            const { value } = await new Luna(uri, params, controller.signal).next();
            controller.abort();
            return value;
        }
        public static subscribe<T>(
            uri: string,
            params: LunaTopicParams = {},
            signal?: AbortSignal,
        ): AsyncIterableIterator<T> {
            return new Luna(uri, { ...params, subscribe: true }, signal);
        }
        private handleAbort() {
            this.end();
            this.bridge.cancel();
            this.signal.removeEventListener('abort', this.handleAbort);
        }
        private handleMessage(serialized: string) {
            this.write(JSON.parse(serialized));
        }
    }
