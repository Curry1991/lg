So I figured out one way how we can keep the display awake and prevent
screensaver, the thing is I thought I found two but I have no idea why
the other method isn't working.. What works is we subscribe to screen
saver requests and NACK them each time:
    function respondToScreenSaverRequest(clientName, timestamp, ack) {
        new LS2Request().send({
            service: 'luna://com.webos.service.tvpower/',
            method: 'power/responseScreenSaverRequest',
            parameters: {clientName, timestamp, ack},
            onSuccess: (res) => {
                console.log('respondToScreenSaverRequest:', res);
            },
        });
    }
    function registerScreenSaverRequest(clientName) {
        new LS2Request().send({
            service: 'luna://com.webos.service.tvpower/',
            method: 'power/registerScreenSaverRequest',
            parameters: {clientName, subscribe: true},
            onSuccess: (res) => {
                console.log('registerScreenSaverRequest:', res);
                if (res.timestamp) {
                    respondToScreenSaverRequest(clientName, res.timestamp, false); // Send NACK.
                }
            },
        });
    }
    registerScreenSaverRequest(appId);
