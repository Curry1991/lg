    readgeneric() {
        # file, offset, count, echo result
        file=""
        [[ "${1}" != "-" ]] && file="if=${1}"
        IFS=' ' data=($(dd ${file} skip=${2} count=${3} status=none bs=1 | tohex ${3}))
        # reverse to get the correct byte order
        rdata=""
        for ((x=${#data[@]}; x>=0; x--)); do
            rdata=${rdata}${data[${x}]}
        done
        if [[ ! -z ${4} ]]; then
            echo $((16#${rdata}))
        elif [[ ${3} -eq 2 ]]; then
            retshort=$((16#${rdata}))
        elif [[ ${3} -eq 4 ]]; then
            retint=$((16#${rdata}))
        elif [[ ${3} -eq 8 ]]; then
            retlong=$((16#${rdata}))
        else
            echo $((16#${rdata}))
        fi
    }

            "\x59\xc3"     # pop rcx; ret
            "\xff\xd0\xc3" # pop r8 ; ret
