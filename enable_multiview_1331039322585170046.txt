    import sys
    import json
    import subprocess
    with open(sys.argv[1]) as f:
      data = json.load(f)
    converted = { f'multiview.{key}' : data[key] for key in data }
    payload = json.dumps({'configs': converted})
    print(payload)
    subprocess.run(['luna-send', '-f', '-n', '1', 'luna://com.webos.service.config/setConfigs', payload])
