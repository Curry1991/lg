    class PackageEntry extends BaseEntry {
        private Struct $hdrStruct;
        public function __construct(Windows $windows, string $name, $value){
            parent::__construct($windows, $name, $value);
            $this->hdrStruct = new Struct(array(
                'name_length' => 'V',
                'unk_int' => 'V'
            ), 8);
        }
        public function packageName(){
            $hdr = $this->hdrStruct->fromData(
                substr($this->entryValue, 0, 8)
            );
            $pkgName = substr($this->entryValue, 8, $hdr['name_length']);
            return $pkgName;
        }
    }

    class ComponentEntryFactory {
        private Windows $windows;
        private Registry $reg;
        private RegistryKey $key;
        public function __construct(Windows $windows, RegistryKey $key){
            $this->windows = $windows;
            $this->reg = $windows->getRegistry();
            $this->key = $key;
        }
        public function entries(){
            foreach($this->key->values() as $valueName){
                $parts = explode('!', $valueName, 2);
                if(count($parts) != 2) continue;
                list($type, $name) = $parts;
                $value = $this->key->read($valueName);
                switch($type){
                    case 'c': // component
                        yield new Entries\ComponentEntry($this->windows, $name);
                    case 'p': // package
                        yield new Entries\PackageEntry($this->windows, $name, $value);
                    case 'f': // file
                        yield new Entries\FileEntry($this->windows, $name);
                    case 'f256': // file sha256
                        yield new Entries\HashEntry($this->windows, $name, $value);
                    case 'i': // unknown (index?)
                    case 's': // unknown (source?)
                        break;
                    default:
                        throw new \InvalidArgumentException("Unsupported type '{$type}'");
                }
            }
        }
    }
