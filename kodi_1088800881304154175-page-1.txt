    #!/bin/sh
    usb_dev='/dev/sda1'
    usb_path='/tmp/usb/sda/sda1'
    [ -b "${usb_dev}" ] || { echo >&2 "error: USB device '${usb_dev}' not found"; exit 1; }
    [ -d "${usb_path}" ] || mkdir -p -- "${usb_path}"
    if ! mountpoint -q -- "${usb_path}"; then
        echo "USB device not mounted; mounting..."
        mount -- "${usb_dev}" "${usb_path}" || { echo >&2 "error: failed to mount USB device '${usb_dev}' at '${usb_path}'"; exit 1; }
    fi
    appid='org.xbmc.kodi'
    app_path="/media/developer/apps/usr/palm/applications/${appid}"
    jail_app_path="/var/palm/jail/${appid}${app_path}"
    bind_mount() {
        [ "${#}" -eq 1 ] || { echo >&2 "error: wrong number of arguments: ${#} but should be 1"; exit 1; }
        usb="${usb_path}/${1}"
        jail="${jail_app_path}/${1}"
        [ -d "${usb}" ] || { echo "creating directory '${usb}'"; mkdir -- "${usb}"; }
        mount --bind -- "${usb}" "${jail}"
    }
    bind_mount '.kodi'
    bind_mount 'addons'
